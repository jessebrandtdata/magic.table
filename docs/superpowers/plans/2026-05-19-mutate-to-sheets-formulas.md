# `mutate()` → Sheets Formulas Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a lazy `sheet_frame` + `mutate()` API that translates tidy expressions into live Google Sheets formulas (`=B4-C4`, …), with an extensible translator dispatch table.

**Architecture:** S3 class `sheet_frame` accumulates `dplyr::mutate()` calls as a list of pending ops without evaluating them. A pure orchestrator `as_sheet_cells()` walks each op's quosures via an AST translator (`walk_expr`), which dispatches every R call against an internal dialect environment that users can extend. `write_sheet()` is a thin shell over `as_sheet_cells()` + `googlesheets4::range_write()`. Existing eager helpers (`make_summary_row`, `write_analytical_table`) stay untouched.

**Tech Stack:** R 4.x, `rlang` (quosure capture + AST predicates), `dplyr` (only the `mutate` generic), `tibble`, `googlesheets4`, `testthat` 3e.

**Spec:** `docs/superpowers/specs/2026-05-19-mutate-to-sheets-formulas-design.md`

```r
#' A lazy sheet-bound data frame
#'
#' Wraps a data frame so that subsequent [mutate()][dplyr::mutate] calls record
#' pending formula columns instead of evaluating them. Materialize the result
#' with [write_sheet()] or inspect what would be written with [as_sheet_cells()].
#'
#' @param df A data frame.
#'
#' @return An object of class `c("sheet_frame", "tbl")`.
#'
#' @examples
#' df <- data.frame(region = c("E", "W"), sales = c(10, 20), cost = c(4, 8))
#' as_sheet_frame(df)
#'
#' @export
as_sheet_frame <- function(df) {
  if (!is.data.frame(df)) {
    stop("`df` must be a data frame.", call. = FALSE)
  }
  structure(
    list(data = tibble::as_tibble(df), ops = list()),
    class = c("sheet_frame", "tbl")
  )
}

# Internal accessors. Not exported.
sheet_frame_data <- function(x) x$data
sheet_frame_ops  <- function(x) x$ops

#' @export
print.sheet_frame <- function(x, ...) {
  cat("<sheet_frame> ", nrow(x$data), " rows x ", ncol(x$data), " cols\n", sep = "")
  cat("  schema: ", paste(names(x$data), collapse = ", "), "\n", sep = "")
  cat("  ", length(x$ops), " pending op", if (length(x$ops) != 1L) "s" else "",
      "\n", sep = "")
  invisible(x)
}
```

```r
# Package-private environment holding the active translator dispatch table.
# Each entry is function(args, ctx) -> character(1), where:
#   args : list of already-translated child strings, in source order
#   ctx  : list with at least `schema` (named char vec: column-name -> letter)
#          and `row_n` (1-based sheet row of the current data cell)
the <- new.env(parent = emptyenv())
the$translators <- list()

#' Return the active Sheets-formula translator dispatch table
#'
#' Each entry is keyed by the R call head it translates (e.g. `"+"`,
#' `"ifelse"`) and is a function `function(args, ctx) -> character(1)`.
#'
#' @return A named list of translator functions.
#' @export
sheet_translators <- function() {
  as.list(the$translators)
}

#' Register or remove a Sheets-formula translator
#'
#' @param name Single non-empty character string. The R call head to
#'   translate (e.g. `"pmax"`, `"paste0"`).
#' @param fn A function `function(args, ctx) -> character(1)` that returns a
#'   formula fragment, or `NULL` to remove the translator.
#'
#' @return Invisibly, the previous translator at `name` (or `NULL`).
#' @export
register_sheet_translator <- function(name, fn) {
  if (!is.character(name) || length(name) != 1L || is.na(name) || !nzchar(name)) {
    stop("`name` must be a single non-empty character string.", call. = FALSE)
  }
  if (!is.null(fn) && !is.function(fn)) {
    stop("`fn` must be a function or NULL.", call. = FALSE)
  }
  prev <- the$translators[[name]]
  if (is.null(fn)) {
    the$translators[[name]] <- NULL
  } else {
    the$translators[[name]] <- fn
  }
  invisible(prev)
}

# ---- v1 defaults: binary arithmetic --------------------------------------

# Helper: build a translator for a left-associative binary infix operator.
# We parenthesize each arg so precedence is preserved when nested expressions
# bubble up. The parens are slightly noisy but correct; readability is fine.
.binop <- function(symbol) {
  force(symbol)
  function(args, ctx) {
    if (length(args) != 2L) {
      stop("operator `", symbol, "` expects 2 arguments, got ", length(args),
           ".", call. = FALSE)
    }
    paste0("(", args[[1]], ")", symbol, "(", args[[2]], ")")
  }
}

.install_defaults <- function() {
  the$translators[["+"]] <- .binop("+")
  the$translators[["-"]] <- function(args, ctx) {
    # `-` is unary or binary; handle both.
    if (length(args) == 1L) return(paste0("-(", args[[1]], ")"))
    if (length(args) == 2L) return(paste0("(", args[[1]], ")-(", args[[2]], ")"))
    stop("operator `-` expects 1 or 2 arguments, got ", length(args), ".",
         call. = FALSE)
  }
  the$translators[["*"]] <- .binop("*")
  the$translators[["/"]] <- .binop("/")
  the$translators[["^"]] <- .binop("^")
}

# Install defaults at package load. .onLoad is picked up automatically.
.onLoad <- function(libname, pkgname) {
  .install_defaults()
}

# Also install at source time so devtools::load_all() / tests work without
# triggering .onLoad twice or losing entries between reloads.
.install_defaults()
```

```r
# Pure AST walker. No I/O, no Sheets calls.
#
# walk_expr(expr, schema, row_n) -> character(1) formula fragment.
#
# - expr:   an unevaluated R expression (typically rlang::quo_get_expr(quosure))
# - schema: named character vector mapping column-name -> column-letter
#           (e.g. c(sales = "B", cost = "C"))
# - row_n:  1-based sheet row of the current data cell
walk_expr <- function(expr, schema, row_n) {
  ctx <- list(schema = schema, row_n = row_n)
  .walk(expr, ctx)
}

.walk <- function(expr, ctx) {
  if (rlang::is_symbol(expr)) {
    return(.translate_symbol(expr, ctx))
  }
  if (rlang::is_call(expr)) {
    return(.translate_call(expr, ctx))
  }
  # Anything else is a literal (numeric, logical, character, or NA).
  .translate_literal(expr, ctx)
}

.translate_symbol <- function(sym, ctx) {
  name <- as.character(sym)
  letter <- ctx$schema[[name]]
  if (is.null(letter) || is.na(letter)) {
    stop("column `", name, "` not found in sheet_frame (have: ",
         paste(names(ctx$schema), collapse = ", "), ").", call. = FALSE)
  }
  paste0(letter, ctx$row_n)
}

.translate_literal <- function(lit, ctx) {
  if (length(lit) != 1L) {
    stop("only scalar literals are supported (got length ", length(lit), ").",
         call. = FALSE)
  }
  if (is.na(lit)) {
    stop("NA literals are not supported; use `is.na(col)` to test for blank cells.",
         call. = FALSE)
  }
  if (is.logical(lit))   return(if (isTRUE(lit)) "TRUE" else "FALSE")
  if (is.numeric(lit))   return(format(lit, scientific = FALSE, trim = TRUE))
  if (is.character(lit)) {
    escaped <- gsub('"', '""', lit, fixed = TRUE)
    return(paste0('"', escaped, '"'))
  }
  stop("unsupported literal type: ", typeof(lit), call. = FALSE)
}

.translate_call <- function(call, ctx) {
  head <- call[[1]]
  if (!rlang::is_symbol(head)) {
    stop("only named function calls are supported (got a complex call head).",
         call. = FALSE)
  }
  name <- as.character(head)
  translator <- the$translators[[name]]
  if (is.null(translator)) {
    stop("no Sheets translator for `", name,
         "`; register one with `register_sheet_translator()`.", call. = FALSE)
  }
  args <- lapply(as.list(call)[-1L], .walk, ctx = ctx)
  translator(args, ctx)
}
```

```r
# Helper: build a translator that emits FN(arg1,arg2,...) for any arity.
.fn_call <- function(sheet_name) {
  force(sheet_name)
  function(args, ctx) {
    paste0(sheet_name, "(", paste(args, collapse = ","), ")")
  }
}

# Helper: a binary infix where the two sides should NOT be re-parenthesized.
# Used for comparisons, whose precedence in Sheets matches R's.
.infix <- function(symbol) {
  force(symbol)
  function(args, ctx) {
    if (length(args) != 2L) {
      stop("operator `", symbol, "` expects 2 arguments, got ", length(args),
           ".", call. = FALSE)
    }
    paste0("(", args[[1]], ")", symbol, "(", args[[2]], ")")
  }
}

.install_defaults <- function() {
  # Arithmetic
  the$translators[["+"]] <- .binop("+")
  the$translators[["-"]] <- function(args, ctx) {
    if (length(args) == 1L) return(paste0("-(", args[[1]], ")"))
    if (length(args) == 2L) return(paste0("(", args[[1]], ")-(", args[[2]], ")"))
    stop("operator `-` expects 1 or 2 arguments, got ", length(args), ".",
         call. = FALSE)
  }
  the$translators[["*"]] <- .binop("*")
  the$translators[["/"]] <- .binop("/")
  the$translators[["^"]] <- .binop("^")

  # Comparisons
  the$translators[["<"]]  <- .infix("<")
  the$translators[["<="]] <- .infix("<=")
  the$translators[[">"]]  <- .infix(">")
  the$translators[[">="]] <- .infix(">=")
  the$translators[["=="]] <- .infix("=")
  the$translators[["!="]] <- .infix("<>")

  # Logical (function-form in Sheets)
  the$translators[["&"]] <- .fn_call("AND")
  the$translators[["|"]] <- .fn_call("OR")
  the$translators[["!"]] <- function(args, ctx) {
    if (length(args) != 1L) {
      stop("operator `!` expects 1 argument, got ", length(args), ".",
           call. = FALSE)
    }
    paste0("NOT(", args[[1]], ")")
  }

  # Conditional
  the$translators[["ifelse"]] <- .fn_call("IF")

  # Curated math/stat
  the$translators[["abs"]]   <- .fn_call("ABS")
  the$translators[["round"]] <- .fn_call("ROUND")
  the$translators[["sqrt"]]  <- .fn_call("SQRT")
  the$translators[["pmin"]]  <- .fn_call("MIN")
  the$translators[["pmax"]]  <- .fn_call("MAX")

  # Predicates
  the$translators[["is.na"]] <- .fn_call("ISBLANK")
}
```

`.binop` from Task 3 is unchanged. The duplicate-looking `.binop`/`.infix` is intentional: `.binop` is the public name from Task 3 and protects arithmetic precedence; `.infix` is identical today but conceptually carries comparison semantics — we keep them separate so we can tighten parens differently later if needed.

(If you find the duplication ugly: collapse `.binop` and `.infix` into one helper. It's two functions because they document intent, not because they need different code today.)

```r
#' Record pending formula columns on a sheet_frame
#'
#' S3 method for [dplyr::mutate()]. Captures the supplied expressions as
#' quosures without evaluating them; translation to Sheets formulas happens
#' at [write_sheet()] / [as_sheet_cells()] time.
#'
#' @param .data A [sheet_frame].
#' @param ... Named expressions, e.g. `margin = sales - cost`.
#'
#' @return The input `sheet_frame` with one additional pending mutate op.
#' @export
#' @importFrom dplyr mutate
#' @method mutate sheet_frame
mutate.sheet_frame <- function(.data, ...) {
  quos <- rlang::enquos(...)
  if (length(quos) == 0L) return(.data)
  nms <- names(quos)
  if (is.null(nms) || any(!nzchar(nms))) {
    stop("all mutate arguments must be named (e.g. `margin = sales - cost`).",
         call. = FALSE)
  }
  .data$ops <- c(.data$ops, list(list(verb = "mutate", cols = quos)))
  .data
}
```

```r
#' Materialize a sheet_frame's pending ops into a cell tibble (pure)
#'
#' Walks every pending [mutate()][dplyr::mutate] op and translates each
#' captured expression into a column of `=`-prefixed
#' [googlesheets4::gs4_formula()] strings, anchored at `start_row`. Returns
#' the resulting tibble without touching Google Sheets.
#'
#' This is the testability lever: end-to-end correctness of the lazy
#' API can be exercised without auth.
#'
#' @param sf A [sheet_frame].
#' @param start_row Single positive integer. The 1-based sheet row where the
#'   header lands; data cells live at rows `start_row + 1` .. `start_row + nrow`.
#'
#' @return A tibble; formula columns hold `gs4_formula()` strings.
#' @export
as_sheet_cells <- function(sf, start_row = 1L) {
  if (!inherits(sf, "sheet_frame")) {
    stop("`sf` must be a sheet_frame.", call. = FALSE)
  }
  if (!is.numeric(start_row) || length(start_row) != 1L ||
      is.na(start_row) || start_row < 1 ||
      start_row != as.integer(start_row)) {
    stop("`start_row` must be a single positive integer.", call. = FALSE)
  }
  start_row <- as.integer(start_row)

  data <- sf$data
  n_rows <- nrow(data)
  data_first_row <- start_row + 1L
  data_rows <- seq.int(data_first_row, length.out = n_rows)

  schema <- stats::setNames(col_letter(seq_along(data)), names(data))

  for (op in sf$ops) {
    if (op$verb != "mutate") {
      stop("unsupported pending op verb: ", op$verb, call. = FALSE)
    }
    for (nm in names(op$cols)) {
      q <- op$cols[[nm]]
      expr <- rlang::quo_get_expr(q)
      formulas <- vapply(data_rows, function(r) {
        paste0("=", walk_expr(expr, schema, row_n = r))
      }, character(1))
      data[[nm]] <- googlesheets4::gs4_formula(formulas)
      schema <- c(schema,
                  stats::setNames(col_letter(length(schema) + 1L), nm))
    }
  }
  tibble::as_tibble(data)
}
```

```r
#' Write a sheet_frame to a Google Sheet as live formulas
#'
#' Materializes `sf` via [as_sheet_cells()] and writes the resulting cell
#' tibble to `sheet` in `ss` starting at `start_row`. Formula columns land
#' as live `=`-prefixed cells; other columns pass through their R values.
#'
#' v1 writes the header row + data rows in one call. It does not add title
#' rows or summary rows — compose with [make_summary_row()] +
#' [googlesheets4::sheet_append()] for richer layouts.
#'
#' @param sf A [sheet_frame].
#' @param ss A Google Sheets identifier accepted by
#'   [googlesheets4::range_write()] (URL, sheet ID, or `dribble`).
#' @param sheet Single non-empty character string. The sheet (tab) name.
#' @param start_row Single positive integer. 1-based row of the header.
#'   Defaults to `1`.
#' @param .writer Function used to perform the write. Defaults to
#'   [googlesheets4::range_write()]. Override in tests.
#'
#' @return The cell tibble that was written, invisibly.
#' @export
write_sheet <- function(sf, ss, sheet, start_row = 1L,
                        .writer = googlesheets4::range_write) {
  if (!inherits(sf, "sheet_frame")) {
    stop("`sf` must be a sheet_frame.", call. = FALSE)
  }
  if (is.null(ss)) {
    stop("`ss` must not be NULL.", call. = FALSE)
  }
  if (!is.character(sheet) || length(sheet) != 1L || is.na(sheet) ||
      !nzchar(sheet)) {
    stop("`sheet` must be a single non-empty character string.", call. = FALSE)
  }
  cells <- as_sheet_cells(sf, start_row = start_row)
  range <- paste0(sheet, "!A", as.integer(start_row))
  .writer(ss, data = cells, range = range)
  invisible(cells)
}
```
